contract BankAccount {
address public owner;
uint256 private balance;
constructor() {
owner = msg.sender;
}
function deposit() public payable {
require(msg.sender==owner,"Only owner can deposite");
require(msg.value > 0, "Deposit amount must be greater than zero");
balance += msg.value;
}
function withdraw() public payable{
require(msg.sender == owner, "Only owner can withdraw");
require(msg.value > 0, "Amount must be greater than zero");
require(msg.value <= balance, "Insufficient balance");
balance -= msg.value;
}
function getBalance() public view returns (uint256) {
return balance;
}
}

# B4: Gradient Descent to find the local minimum of y = (x + 3)^2, starting at x = 2
# dy/dx = 2(x + 3)   update rule: x = x - lr * dy/dx   (true minimum: x = -3, y = 0)

import numpy as np
import matplotlib.pyplot as plt

def f(x):
    return (x + 3) ** 2

def grad(x):
    return 2 * (x + 3)

x, lr, tol, max_iter = 2.0, 0.1, 1e-6, 1000 
path = [x]

print("Iter        x            y")
for i in range(1, max_iter + 1):
    step = lr * grad(x)
    x = x - step
    path.append(x)
    if i <= 10 or abs(step) < tol:
        print(f"{i:4d}   {x:10.6f}   {f(x):10.6f}")
    if abs(step) < tol:          # converged
        break

print(f"\nConverged in {i} iterations: x = {x:.6f}, y = {f(x):.8f}")

xs = np.linspace(-8, 4, 200)
plt.plot(xs, f(xs), label="y = (x+3)^2")
plt.scatter(path, [f(p) for p in path], color="red", s=15, label="GD steps")
plt.xlabel("x")
plt.ylabel("y")
plt.title("Gradient Descent from x = 2")
plt.legend()
plt.show()

# B5: K-Means clustering on sales_data_sample.csv; choose K using the Elbow method
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# File is not UTF-8 -> latin1 encoding needed
# df = pd.read_csv(Path(__file__).parent / "sales_data_sample.csv", encoding="latin1")
df = pd.read_csv("sales_data_sample.csv", encoding="latin1")
print("Shape:", df.shape)
features = ["QUANTITYORDERED", "PRICEEACH", "SALES", "MSRP"]
X = StandardScaler().fit_transform(df[features])   # scale so SALES doesn't dominate

# Elbow method: inertia (WCSS) for K = 1..10
wcss = []
for k in range(1, 11):
    km = KMeans(n_clusters=k, n_init=10, random_state=42).fit(X)
    wcss.append(km.inertia_)
    print(f"K={k:2d}  WCSS={km.inertia_:.1f}")

plt.plot(range(1, 11), wcss, marker="o")
plt.xlabel("Number of clusters K")
plt.ylabel("WCSS (inertia)")
plt.title("Elbow Method")
plt.show()

# K chosen where the curve bends (the "elbow")
k = 3
km = KMeans(n_clusters=k, n_init=10, random_state=42).fit(X)
df["Cluster"] = km.labels_

print(f"\nK = {k}\nCluster sizes:\n", df["Cluster"].value_counts().sort_index())
print("\nCluster means:\n", df.groupby("Cluster")[features].mean().round(2))

plt.scatter(df["QUANTITYORDERED"], df["SALES"], c=df["Cluster"], cmap="viridis", s=10)
plt.xlabel("Quantity Ordered")
plt.ylabel("Sales")
plt.title(f"K-Means clusters (K = {k})")
plt.show()

struct node{
    char ch;
    int freq;
    node *left,*right;
    node(char c,int f){
        ch=c;
        freq=f;
        left=right=nullptr;
    }
};
struct compare{
    bool operator()(node*a,node*b){
        return a->freq>b->freq;
    }
};
void print(node*root,string code){
    if(root==nullptr)return;
    if(!root->left && !root->right){
        cout<<root->ch<<" : "<<code<<endl;
        return;
    }
    print(root->left,code+"0");
    print (root->right,code+"1");

}
void huffmanEncoding(char ch[],int freq[],int n){
    priority_queue<node*,vector<node*>,compare>pq;

    for(int i=0;i<n;i++){
        pq.push(new node(ch[i],freq[i]));
    }
    while(pq.size()>1){
        node*left=pq.top();
        pq.pop();
        node*right=pq.top();
        pq.pop();
        node* newnode= new node('$',left->freq+right->freq);
        newnode->left=left;
        newnode->right=right;
        pq.push(newnode);
    }
    cout<<"\nHuffman Codes:\n";
    print(pq.top(),"");
}

#include<bits/stdc++.h>
using namespace std;

int knapsack(int W, int wt[], int val[], int n) {

    int dp[n + 1][W + 1];

    // Initialize DP table
    for (int i = 0; i <= n; i++) {
        for (int w = 0; w <= W; w++) {

            if (i == 0 || w == 0) {
                dp[i][w] = 0;
            }

            else if (wt[i - 1] <= w) {
                dp[i][w] = max(
                    val[i - 1] + dp[i - 1][w - wt[i - 1]],
                    dp[i - 1][w]
                );
            }

            else {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    return dp[n][W];
}

int board[20][20];

// Check whether placing a queen is safe
bool isSafe(int row, int col, int n) {

    // Check same column (above)
    for (int i = 0; i < n; i++) {
        if(i==row)continue;
        if (board[i][col] == 1)
            return false;
    }

    // Check upper-left diagonal
    for (int i = row - 1, j = col - 1;
         i >= 0 && j >= 0; i--, j--) {

        if (board[i][j] == 1)
            return false;
    }

    // Check upper-right diagonal
    for (int i = row - 1, j = col + 1;
         i >= 0 && j < n; i--, j++) {

        if (board[i][j] == 1)
            return false;
    }
    //check down left diagonal
        for (int i = row +1, j = col - 1;
         i <n && j >= 0; i++, j--) {

        if (board[i][j] == 1)
            return false;
    }

    // Check down-right diagonal
    for (int i = row +1, j = col + 1;
         i<n && j < n; i++, j++) {

        if (board[i][j] == 1)
            return false;
    }

    return true;
}

// Solve N-Queens using backtracking
bool solveNQueens(int row, int n, int fixedRow) {

    // All rows completed
    if (row == n)
        return true;

    // Skip the row containing the first queen
    if (row == fixedRow)
        return solveNQueens(row + 1, n, fixedRow);

    for (int col = 0; col < n; col++) {

        if (isSafe(row, col, n)) {

            board[row][col] = 1;

            if (solveNQueens(row + 1, n, fixedRow))
                return true;

            // Backtrack
            board[row][col] = 0;
        }
    }

    return false;
}

    }
}

#include <bits/stdc++.h>
using namespace std;   

int partitionDeterministic(int arr[], int s, int e) {

    int pivot = arr[s];

    int cnt = 0;

    for(int i = s + 1; i <= e; i++) {
        if(arr[i] <= pivot) {
            cnt++;
        }
    }

    int pivotIndex = s + cnt;
    swap(arr[pivotIndex], arr[s]);

    int i = s, j = e;

    while(i < pivotIndex && j > pivotIndex) {

        while(i < pivotIndex && arr[i] <= pivot) {
            i++;
        }

        while(j > pivotIndex && arr[j] > pivot) {
            j--;
        }

        if(i < pivotIndex && j > pivotIndex) {
            swap(arr[i++], arr[j--]);
        }
    }

    return pivotIndex;
}

// Randomized Partition
int partitionRandomized(int arr[], int low, int high) {
    int randomIndex = low + rand() % (high - low + 1);

    swap(arr[randomIndex], arr[low]);

    return partitionDeterministic(arr, low, high);
}

// Deterministic Quick Sort
void quickSortDeterministic(int arr[], int low, int high) {
    if (low < high) {
        int pivot = partitionDeterministic(arr, low, high);

        quickSortDeterministic(arr, low, pivot - 1);
        quickSortDeterministic(arr, pivot + 1, high);
    }
}

// Randomized Quick Sort
void quickSortRandomized(int arr[], int low, int high) {
    if (low < high) {
        int pivot = partitionRandomized(arr, low, high);

        quickSortRandomized(arr, low, pivot - 1);
        quickSortRandomized(arr, pivot + 1, high);
    }
}
